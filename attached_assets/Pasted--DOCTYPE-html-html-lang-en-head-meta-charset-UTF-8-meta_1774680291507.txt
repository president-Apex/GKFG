  function goStep(n) {
    // hide all steps
    document.querySelectorAll('.form-step').forEach(s => s.classList.remove('active'));
    // show target step
    document.getElementById('step-' + n).classList.add('active');
    // update progress
    for (let i = 1; i <= 3; i++) {
      const p = document.getElementById('prog-' + i);
      p.classList.remove('active', 'done');
      if (i < n) p.classList.add('done');
      if (i === n) p.classList.add('active');
    }
    // scroll form into view on mobile
    document.getElementById('form').scrollIntoView({ behavior: 'smooth', block: 'start' });
  }

  function selectService(el) {
    document.querySelectorAll('.service-option').forEach(o => o.classList.remove('selected'));
    el.classList.add('selected');
  }

  function submitForm() {
    // Basic validation
    const fname = document.getElementById('fname').value.trim();
    const phone = document.getElementById('phone').value.trim();
    const email = document.getElementById('email').value.trim();
    if (!fname || !phone || !email) {
      alert('Please fill in your name, phone, and email before submitting.');
      goStep(1);
      return;
    }
    // Hide all steps, show success
    document.querySelectorAll('.form-step').forEach(s => s.classList.remove('active'));
    document.querySelectorAll('.progress-step').forEach(p => { p.classList.remove('active'); p.classList.add('done'); });
    const s = document.getElementById('success-state');
    s.classList.add('visible');
    document.getElementById('form').scrollIntoView({ behavior: 'smooth', block: 'start' });
  }

  // Auto-select service options on click anywhere in the card
  document.querySelectorAll('.service-option').forEach(opt => {
    opt.addEventListener('click', function() {
      document.querySelectorAll('.service-option').forEach(o => o.classList.remove('selected'));
      this.classList.add('selected');
    });
  });
